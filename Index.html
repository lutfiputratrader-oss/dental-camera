  <script>

    const video = document.getElementById("video");

    const canvas = document.getElementById("canvas");

    const ctx = canvas.getContext("2d");

    const viewType = document.getElementById("viewType");

    const startCameraBtn = document.getElementById("startCamera");

    const switchCameraBtn = document.getElementById("switchCamera");

    const captureBtn = document.getElementById("capture");

    const downloadBtn = document.getElementById("download");

    const result = document.getElementById("photoResult");

    const preview = document.getElementById("preview");


    const frame = document.querySelector(".frame");

    const vertical = document.querySelector(".vertical");

    const horizontal = document.querySelector(".horizontal");

    const occlusalLine = document.querySelector(".occlusal-line");

    const arch = document.querySelector(".arch");

    const caption = document.querySelector(".caption");

    const templateName = document.getElementById("templateName");

    const levelStatus = document.getElementById("levelStatus");


    let stream = null;

    let useBackCamera = true;

    let photoData = null;


    const templates = {

      anterior: {

        name: "Anterior / Frontal",

        text: "Sejajarkan midline dengan garis vertikal dan bidang oklusal dengan garis horizontal.",

        show: ["frame", "vertical", "horizontal"]

      },

      right: {

        name: "Buccal Kanan",

        text: "Pastikan oklusi dan area posterior terlihat; jaga bidang oklusal sejajar.",

        show: ["frame", "horizontal"]

      },

      left: {

        name: "Buccal Kiri",

        text: "Pastikan oklusi dan area posterior terlihat; jaga bidang oklusal sejajar.",

        show: ["frame", "horizontal"]

      },

      upper: {

        name: "Oklusal Maksila",

        text: "Gunakan mirror; tempatkan midline dan lengkung rahang pada template.",

        show: ["vertical", "occlusal", "arch"]

      },

      lower: {

        name: "Oklusal Mandibula",

        text: "Gunakan mirror; tempatkan midline dan lengkung rahang pada template.",

        show: ["vertical", "occlusal", "arch", "lowerArch"]

      }

    };


    function updateGuide() {

      const t = templates[viewType.value];


      frame.style.display = "none";

      vertical.style.display = "none";

      horizontal.style.display = "none";

      occlusalLine.style.display = "none";

      arch.style.display = "none";

      arch.classList.remove("lower");


      if (t.show.includes("frame")) frame.style.display = "block";

      if (t.show.includes("vertical")) vertical.style.display = "block";

      if (t.show.includes("horizontal")) horizontal.style.display = "block";

      if (t.show.includes("occlusal")) occlusalLine.style.display = "block";

      if (t.show.includes("arch")) arch.style.display = "block";

      if (t.show.includes("lowerArch")) arch.classList.add("lower");


      caption.style.display = "block";

      caption.textContent = t.text;

      templateName.textContent = t.name;

    }


    async function startCamera() {

      if (stream) {

        stream.getTracks().forEach(track => track.stop());

      }


      try {

        stream = await navigator.mediaDevices.getUserMedia({

          video: {

            facingMode: useBackCamera ? { ideal: "environment" } : "user",

            width: { ideal: 1920 },

            height: { ideal: 1080 }

          },

          audio: false

        });


        video.srcObject = stream;

        startCameraBtn.textContent = "Kamera Aktif";

      } catch (error) {

        alert("Kamera tidak dapat diakses. Pastikan izin kamera aktif dan halaman dibuka melalui HTTPS atau localhost.");

        console.error(error);

      }

    }


    function capturePhoto() {

      if (!video.videoWidth) {

        alert("Aktifkan kamera terlebih dahulu.");

        return;

      }


      canvas.width = video.videoWidth;

      canvas.height = video.videoHeight;


      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);


      // Watermark nama template pada hasil dokumentasi

      ctx.fillStyle = "rgba(0, 0, 0, 0.55)";

      ctx.fillRect(20, canvas.height - 95, 430, 65);


      ctx.fillStyle = "#ffffff";

      ctx.font = "bold 28px Arial";

      ctx.fillText(`Dental View: ${templates[viewType.value].name}`, 35, canvas.height - 55);


      photoData = canvas.toDataURL("image/jpeg", 0.95);

      result.src = photoData;

      preview.style.display = "block";

    }


    function downloadPhoto() {

      if (!photoData) {

        alert("Ambil foto terlebih dahulu.");

        return;

      }


      const link = document.createElement("a");

      const now = new Date().toISOString().replace(/[:.]/g, "-");

      link.download = `dental-${viewType.value}-${now}.jpg`;

      link.href = photoData;

      link.click();

    }


    async function enableLevel() {

      if (typeof DeviceOrientationEvent !== "undefined" &&

          typeof DeviceOrientationEvent.requestPermission === "function") {

        try {

          await DeviceOrientationEvent.requestPermission();

        } catch (e) {

          console.log("Sensor orientasi tidak diizinkan.");

        }

      }

    }


    window.addEventListener("deviceorientation", (event) => {

      if (event.gamma === null) return;


      const tilt = Math.abs(event.gamma);


      if (tilt < 4) {

        levelStatus.innerHTML = 'Kemiringan: <strong>Sejajar</strong>';

      } else {

        levelStatus.innerHTML = `Kemiringan: <strong>${tilt.toFixed(0)}°</strong>`;

      }

    });


    startCameraBtn.addEventListener("click", async () => {

      await enableLevel();

      startCamera();

    });


    switchCameraBtn.addEventListener("click", () => {

      useBackCamera = !useBackCamera;

      startCamera();

    });


    captureBtn.addEventListener("click", capturePhoto);

    downloadBtn.addEventListener("click", downloadPhoto);

    viewType.addEventListener("change", updateGuide);


    updateGuide();

  </script>
